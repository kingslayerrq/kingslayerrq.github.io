No.05 // Software // 2023Hand-coded
nyuenc.c · worker loop, debug comments trimmed
void *worker(void *arg)
{
    threadpool_t *pool = (threadpool_t *)arg;
    while (1)
    {
        pthread_mutex_lock(&(pool->qlock));
        while (pool->taskQueueSize == 0 && pool->encodedNum < resCount)
        {
            pthread_cond_wait(&(pool->task_not_empty), &(pool->qlock));
        }
        if (pool->encodedNum >= resCount)
        {
            pthread_mutex_unlock(&(pool->qlock));
            pthread_exit(NULL);
        }
        //  Accept a Task
        thread_worker *t = pool->taskQueue;
        pool->taskQueue = t->next;
        pool->taskQueueSize--;
        pool->encodedNum++;
        pthread_mutex_unlock(&(pool->qlock));
        // Runs the Task
        t->res = t->func(t->arg, t->chunkSize);
